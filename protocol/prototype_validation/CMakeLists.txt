cmake_minimum_required(VERSION 3.16)
project(prototype_validation_v2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
add_executable(prototype_validation_v2 prototype_validation_v2.cpp)
target_link_libraries(prototype_validation_v2 PRIVATE OpenSSL::Crypto)
if(NOT MSVC)
  target_compile_options(prototype_validation_v2 PRIVATE -O2 -Wall -Wextra -Wpedantic)
endif()
